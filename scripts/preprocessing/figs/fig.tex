\documentclass{article}
\usepackage{tikz}
\usepackage[active,tightpage]{preview}
\PreviewEnvironment{tikzpicture}
\usepackage{amsmath}
\usetikzlibrary{positioning, shapes.geometric}

\begin{document}

\begin{tikzpicture}[
  node distance=0.8cm and 1.5cm,
  layer/.style={draw, rectangle, minimum width=3.8cm, minimum height=1cm},
  conv/.style={layer, fill=blue!10},
  relu/.style={layer, fill=green!10},
  pool/.style={layer, fill=orange!15},
  flat/.style={layer, fill=yellow!20},
  fc/.style={layer, fill=red!10}
]

% Input
\node[layer] (input) {Input Image (2×64×64)};

% Convolutional layers
\node[conv, below=of input] (conv1) {Conv2D(2→16, 3×3)};
\node[relu, below=of conv1] (relu1) {ReLU};
\node[pool, below=of relu1] (pool1) {MaxPool2D(2×2)};

\node[conv, below=of pool1] (conv2) {Conv2D(16→32, 3×3)};
\node[relu, below=of conv2] (relu2) {ReLU};
\node[pool, below=of relu2] (pool2) {MaxPool2D(2×2)};

\node[conv, below=of pool2] (conv3) {Conv2D(32→64, 3×3)};
\node[relu, below=of conv3] (relu3) {ReLU};
\node[pool, below=of relu3] (pool3) {MaxPool2D(2×2)};

% Flatten
\node[flat, below=of pool3] (flatten) {Flatten (64×8×8 = 4096)};

% Fully connected layers
\node[fc, below=of flatten] (fc1) {Linear(4096→128)};
\node[relu, below=of fc1] (relu4) {ReLU};
\node[fc, below=of relu4] (fc2) {Linear(128→4)};

% Output
\node[layer, below=of fc2] (output) {Output: [x\_min, x\_max, y\_min, y\_max]};

% Arrows
\foreach \i/\j in {input/conv1, conv1/relu1, relu1/pool1, pool1/conv2, conv2/relu2,
                   relu2/pool2, pool2/conv3, conv3/relu3, relu3/pool3, pool3/flatten,
                   flatten/fc1, fc1/relu4, relu4/fc2, fc2/output}
  \draw[->] (\i) -- (\j);

\end{tikzpicture}

\end{document}
